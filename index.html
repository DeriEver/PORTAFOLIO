PHP

Lógica de conexión y CRUD:

<?php
$conexion = new mysqli("localhost", "root", "", "base_examen");

$i1 = $_POST['i1'] ?? '';
$i2 = $_POST['i2'] ?? '';
$i3 = $_POST['i3'] ?? '';
$i4 = $_POST['i4'] ?? '';
$btn = $_POST['btn'] ?? '';


if ($btn == 'R') {
    $conexion->query("INSERT INTO tabla VALUES ('$i1', '$i2', '$i3', '$i4')");
    echo "Registrado";
}

if ($btn == 'M') {
    $conexion->query("UPDATE tabla SET c2='$i2', c3='$i3', c4='$i4' WHERE c1='$i1'");
    echo "Modificado";
}

if ($btn == 'E') {
    $conexion->query("DELETE FROM tabla WHERE c1='$i1'");
    echo "Eliminado";
}

if ($btn == 'C') {
    $resultado = $conexion->query("SELECT * FROM tabla WHERE c1='$i1'");
    foreach ($resultado as $fila) {
        echo "Encontrado: " . $fila['c2'] . " | " . $fila['c3'] . " | " . $fila['c4'];
    }
}
?>
